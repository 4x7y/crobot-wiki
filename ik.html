Inverse Kinematics

Surface-Curve Contact Constraint

Value

\[\begin{split}\begin{align*} ^{w}p_e &= {}^{w}s(u_s, v_s) - {}^{w}c(u_c) \\[1em] &= \Big( R_{ws} {}^{b}s(u_s, v_s) + {}^{w}p_{s} \Big) - \Big( R_{wc} {}^{b}c(u_c) + {}^{w}p_{c} \Big) \\[2em] \xi &= {}^w \hat{n}_s \, \cdot \, {}^w \hat{t}_c \\[1em] &= \Big( R_{ws} {}^b\hat{n}_s(u_s, v_s) \Big) \, \cdot \, \Big( R_{wc}\, {}^b\hat{t}_c(u_c) \Big) \end{align*}\end{split}\]

Bounds

\[\begin{split}\begin{gather*} -\varepsilon \leq {}^{w}p_e \leq \varepsilon \\[1em] -\varepsilon \leq \xi \leq \varepsilon \end{gather*}\end{split}\]

Jacobian

\[\begin{split}\begin{align*} \frac{\partial\, {}^w p_e}{\partial q} &= -\Big[R_{ws} {}^{b}s(u_s, v_s) \Big]_\times J_{r, s}(q) + J_{p, s}(q) + \Big[R_{wc} {}^{b}c(u_c) \Big]_\times J_{r, c}(q) - J_{p, c}(q) \\[1em] \frac{\partial\, {}^w p_e}{\partial u_s} &= R_{ws} \frac{\partial\, {}^{b}s(u_s, v_s)}{\partial u_s} \\[1em] \frac{\partial\, {}^w p_e}{\partial v_s} &= R_{ws} \frac{\partial\, {}^{b}s(u_s, v_s)}{\partial v_s} \\[1em] \frac{\partial\, {}^w p_e}{\partial u_c} &= -R_{wc} \frac{\partial\,{}^{b}c(u_c)}{\partial u_c} \\[1em] \frac{\partial \xi}{\partial q} &= -\Big( R_{wc}\, {}^b\hat{t}_c(u_c) \Big)^\top \Big[ R_{ws} {}^b\hat{n}_s(u_s, v_s) \Big]_\times J_{r,s} \\[1em] &\,\, \quad -\Big( R_{ws} {}^b\hat{n}_s(u_s, v_s) \Big)^\top \Big[ R_{wc}\, {}^{b} c(u_c) \Big]_\times J_{r, c} \\[1em] \frac{\partial \xi}{\partial u_s} &= \Big( R_{wc}\, {}^b\hat{t}_c(u_c) \Big)^\top R_{ws} \frac{\partial \, {}^b \hat{n}_s(u_s, v_s)}{\partial u_s} \\[1em] \frac{\partial \xi}{\partial v_s} &= \Big( R_{wc}\, {}^b\hat{t}_c(u_c) \Big)^\top R_{ws} \frac{\partial \, {}^b \hat{n}_s(u_s, v_s)}{\partial v_s} \\[1em] \frac{\partial \xi}{\partial u_c} &= \Big( R_{ws} {}^b\hat{n}_s(u_s, v_s) \Big)^\top R_{wc} \frac{\partial \, {}^b\hat{t}_c(u_c)}{\partial u_c} \end{align*}\end{split}\]

Bounding Box Constraint

Add a bounding box constraint on the position of one body (end-effector)relative to a target bounding box fixed on another body (the target body). This constraint is satisfied if the position of the end-effector bpos relative to the target bounding box center remains within the Bounds specified.

  • \(p\): A point fixed to the end-effector frame.

  • \(p_e\): Origin of the end-effector frame.

  • \(p_t\): Origin of the target frame.

  • \(p_b\): Center of the bounding box frame.

Value

\[^{b}p' = T_{tb}^{-1} T_{wt}^{-1} T_{we} {}^{e}p'\]

Bounds

\[{}^{b}p_{lb} \leq {}^{b}p \leq {}^{b}p_{ub}\]

Jacobian

\[\begin{align*} {}^{b}p &= R_{tb}^\top \left[ R_{wt}^\top {}^{w}p - R_{wt}^\top {}^{w}p_t \right] - R_{tb}^\top {}^{t}p_b \end{align*}\]
\[\begin{split}\begin{align*} {}^{b}\dot{p} &= R_{tb}^\top \left[ \dot{R}_{wt}^\top {}^{w}p + R_{wt}^\top {}^{w}\dot{p} - \dot{R}_{wt}^\top {}^{w}p_t - R_{wt}^\top {}^{w}\dot{p}_t \right] \\[1em] &= R_{tb}^\top \left[ \dot{R}_{wt}^\top \big({}^{w}p - {}^{w}p_t \big) + R_{wt}^\top {}^{w}\dot{p} - R_{wt}^\top {}^{w}\dot{p}_t \right] \\[1em] &= R_{tb}^\top \Big[ \big(R_{wt} [\omega_t]_\times \big)^\top \big({}^{w}p - {}^{w}p_t \big) + R_{wt}^\top {}^{w}\dot{p} - R_{wt}^\top {}^{w}\dot{p}_t \Big] \\[1em] &= R_{tb}^\top \Big[ [\omega_t]_\times^\top R_{wt}^\top \big({}^{w}p - {}^{w}p_t \big) + R_{wt}^\top {}^{w}\dot{p} - R_{wt}^\top {}^{w}\dot{p}_t \Big] \\[1em] &= R_{tb}^\top \Big[ -[\omega_t]_\times R_{wt}^\top \big({}^{w}p - {}^{w}p_t \big) + R_{wt}^\top {}^{w}\dot{p} - R_{wt}^\top {}^{w}\dot{p}_t \Big] \\[1em] &= R_{tb}^\top \Big[ \Big[R_{wt}^\top \big({}^{w}p - {}^{w}p_t \big) \Big]_\times \omega_t + R_{wt}^\top {}^{w}\dot{p} - R_{wt}^\top {}^{w}\dot{p}_t \Big] \\[1em] &= R_{tb}^\top \Big[ \Big[R_{wt}^\top \big({}^{w}p - {}^{w}p_t \big) \Big]_\times J_{r, t} \,\dot{q} + R_{wt}^\top J_{p}\,\dot{q} - R_{wt}^\top J_{p, t} \,\dot{q} \Big] \\[1em] &= R_{tb}^\top \Big[ \Big[R_{wt}^\top \big({}^{w}p - {}^{w}p_t \big) \Big]_\times J_{r, t} + R_{wt}^\top J_{p} - R_{wt}^\top J_{p, t} \Big] \,\dot{q} \end{align*}\end{split}\]
\[\frac{\partial\, {}^b p}{\partial \, q} = R_{tb}^\top \Big[ \Big[R_{wt}^\top \big({}^{w}p - {}^{w}p_t \big) \Big]_\times J_{r, t} + R_{wt}^\top J_{p} - R_{wt}^\top J_{p, t} \Big]\]
